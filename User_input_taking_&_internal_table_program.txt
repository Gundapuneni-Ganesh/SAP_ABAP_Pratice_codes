     emp_name TYPE string.

*-------------------------------------------------------------------------
* Multiple Fields → TABLE OF structure

TYPES: BEGIN OF ty_student,
         id   TYPE i,
         name TYPE string,
         marks TYPE p DECIMALS 2,
       END OF ty_student.

DATA lt_students TYPE TABLE OF ty_student.

APPEND VALUE ty_student(
  id = 1
  name = 'Ravi'
  marks = '85.50'
) TO lt_students.

APPEND VALUE ty_student(
  id = 2
  name = 'Sita'
  marks = '90.00'
) TO lt_students.

LOOP AT lt_students INTO DATA(ls_student).
  WRITE: / ls_student-id, ls_student-name, ls_student-marks.
ENDLOOP.
