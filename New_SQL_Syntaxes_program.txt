DATA(lv_min_price) = 300.
DATA(lv_year) = '2024'.

SELECT
    " 1. Regular columns
    carrid,
    connid,
    fldate,
    price,

    " 2. Literals
    'TICKET' AS document_type,
    1 AS version,

    " 3. Host variables (@ prefix)
    @lv_year AS report_year,
    @sy-uname AS created_by,

    " 4. Arithmetic expressions
    price * 1.19 AS price_with_vat,
    price DIV 100 AS hundred_blocks,
    price MOD 10 AS last_digit,
    ROUND( price * 0.85, 2 ) AS discounted,

    " 5. CASE expression
    CASE
        WHEN price > 800 THEN 'Premium'
        WHEN price > 500 THEN 'Standard'
        WHEN price > 0   THEN 'Budget'
        ELSE 'Error'
    END AS class,

    " 6. LET for complex calculations
    LET
        vat = price * 0.19,
        fee = CASE
                  WHEN carrid = 'LH' THEN 50
                  ELSE 25
              END,
        total = price + vat + fee,
        tax_rate = ( vat / total ) * 100
    IN
        vat AS vat_amount,
        fee AS service_fee,
        total AS total_price,
        tax_rate AS vat_percentage,

    " 7. Aggregate with window function
    AVG( price ) OVER( PARTITION BY carrid ) AS carrier_avg_price

FROM sflight
WHERE
    price > @lv_min_price
    AND fldate BETWEEN @lv_year && '0101' AND @lv_year && '1231'

INTO TABLE @DATA(lt_complete).

" Display results
LOOP AT lt_complete INTO DATA(ls_result).
  WRITE: / |Flight: { ls_result-carrid } { ls_result-connid }|,
         / |  Price: { ls_result-price }, Class: { ls_result-class }|,
         / |  Total with tax & fee: { ls_result-total_price }|,
         / |  VAT: { ls_result-vat_percentage }%|,
         / |  Carrier avg price: { ls_result-carrier_avg_price }|,
         / |  -------------------|.
ENDLOOP.
